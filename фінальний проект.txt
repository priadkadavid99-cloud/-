import tkinter as tk
from tkinter import messagebox

window = tk.Tk()
window.title("Хрестики-нулики")
window.config(bg="#2c5942")

hod = "X"
knopky = []

try:
    kartynka = tk.PhotoImage(file="гра.png").subsample(x=5, y=5)
    tk.Label(window, image=kartynka, bg="#3e472b").pack(pady=5)
except tk.TclError:
    tk.Label(window, text="Додай image.png у папку", bg="#9b9e96").pack(pady=5)

nadpys = tk.Label(window, text="Хід: X", font=("Arial", 14), bg="#ffffff")
nadpys.pack(pady=5)


def perevirka_peremohy():
    lines = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8),
        (0, 3, 6), (1, 4, 7), (2, 5, 8),
        (0, 4, 8), (2, 4, 6),
    ]
    for a, b, c in lines:
        t1 = knopky[a]["text"]
        t2 = knopky[b]["text"]
        t3 = knopky[c]["text"]
        if t1 != "" and t1 == t2 and t2 == t3:
            return t1
    return ""


def chy_nichya():
    for k in knopky:
        if k["text"] == "":
            return False
    return True


def zminyty_hod():
    global hod
    if hod == "X":
        hod = "O"
    else:
        hod = "X"
    nadpys["text"] = "Хід: " + hod


def natysnuly(nomer):
    if knopky[nomer]["text"] != "":
        return

    knopky[nomer]["text"] = hod
    peremozhec = perevirka_peremohy()

    if peremozhec != "":
        messagebox.showinfo("Перемога", "Виграв " + peremozhec)
        nova_gra()
        return

    if chy_nichya():
        messagebox.showinfo("Нічия", "Нічия!")
        nova_gra()
        return

    zminyty_hod()


def nova_gra(event=None):
    global hod
    hod = "X"
    nadpys["text"] = "Хід: X"
    for k in knopky:
        k["text"] = ""


pole = tk.Frame(window, bg="lightgray")
pole.pack()

for i in range(9):
    k = tk.Button(pole, text="", width=5, height=2, font=("Arial", 20), command=lambda n=i: natysnuly(n))
    k.grid(row=i // 3, column=i % 3, padx=2, pady=2)
    knopky.append(k)

tk.Button(window, text="Нова гра", font=("Arial", 12), command=nova_gra).pack(pady=8)

window.bind("r", nova_gra)
window.bind("R", nova_gra)

window.mainloop()
